SELECT Suffixes.SpellID, Suffixes.Name, Spells.SpellID, Spells.Name FROM (SELECT Spells.SpellID, Spells.Name, TRIM(SUBSTR(Spells.Name, INSTR(Spells.Name, "'") + 2)) AS Suffix FROM Spells) AS Suffixes INNER JOIN Spells ON Suffixes.Suffix = Spells.Name AND Suffixes.Name <> Spells.Name
